import matplotlib.pyplot as plt

import numpy as np

# ── Constantes compartidas ──────────────────────────────────────────
C, L, EPS = 2.0, 1.5, 0.95
D1, D2    = 0.40, 0.58          # δ₁ (izq) < δ₂ (der)
D         = D1                   # δ = min(δ₁, δ₂)

BLUE_Z = '#90c8f0'
GOLD_Z = '#fffde7'
RED_F  = '#c0392b'

def _f(x):
    return L + np.sin(2.0 * (x - C))

def _base(ax, xlo, xhi):
    ylo, yhi = 0.25, 2.75
    ax.set_xlim(xlo, xhi); ax.set_ylim(ylo, yhi)
    ax.set_xticks([]); ax.set_yticks([])
    for sp in ax.spines.values():
        sp.set_linewidth(0.5)

    ax.axhspan(L + EPS, yhi,  alpha=0.40, color=BLUE_Z, zorder=0)
    ax.axhspan(ylo, L - EPS,  alpha=0.40, color=BLUE_Z, zorder=0)

    for y, col in [(L+EPS,'steelblue'),(L,'navy'),(L-EPS,'steelblue')]:
        ax.axhline(y, color=col, ls='--', lw=0.9, zorder=1)
    ax.axvline(C, color='gray', ls='--', lw=0.7, zorder=1)

    xs = np.linspace(xlo, xhi, 800)
    ys = _f(xs)
    mask = np.abs(xs - C) > 0.011
    ax.plot(xs[mask], ys[mask], color=RED_F, lw=1.5, zorder=2)
    ax.plot(C, L, 'o', ms=6.5, mfc='white', mec=RED_F, mew=1.8, zorder=4)

def _zona(ax, xlo, xhi):
    mx = (xlo + xhi) / 2
    ax.text(mx, L+EPS+0.11, 'Zona prohibida',
            ha='center', va='bottom', fontsize=8.2, color='steelblue', fontweight='bold')
    ax.text(mx, L-EPS-0.11, 'Zona prohibida',
            ha='center', va='top',    fontsize=8.2, color='steelblue', fontweight='bold')

def _ylabels_box(ax, xlo, xhi, lbl):

# ── Constantes compartidas ──────────────────────────────────────────
C, L, EPS = 2.0, 1.5, 0.95
D1, D2    = 0.40, 0.58          # δ₁ (izq) < δ₂ (der)
D         = D1                   # δ = min(δ₁, δ₂)

BLUE_Z = '#90c8f0'
GOLD_Z = '#fffde7'
RED_F  = '#c0392b'
NAVY   = 'navy'
GOLD   = 'goldenrod'
GRAY   = 'gray'

def _f_unif(x):
    """Función que se aproxima a L desde ambos lados"""
    return L + 0.6 * np.sin(2.5 * (x - C)) * np.exp(-0.5 * (x - C)**2)

def _base_unif(ax, xlo, xhi):
    ylo, yhi = 0.25, 2.75
    ax.set_xlim(xlo, xhi)
    ax.set_ylim(ylo, yhi)
    ax.set_xticks([])
    ax.set_yticks([])
    for sp in ax.spines.values():
        sp.set_linewidth(0.5)
    
    # Líneas horizontales de referencia
    ax.axhline(L + EPS, color='steelblue', ls='--', lw=0.9, zorder=1)
    ax.axhline(L, color=NAVY, ls='-', lw=0.9, zorder=1)
    ax.axhline(L - EPS, color='steelblue', ls='--', lw=0.9, zorder=1)
    
    # Línea vertical en c
    ax.axvline(C, color=GRAY, ls='--', lw=0.7, zorder=1)
    
    # Función
    xs = np.linspace(xlo, xhi, 800)
    ys = _f_unif(xs)
    mask = np.abs(xs - C) > 0.015
    ax.plot(xs[mask], ys[mask], color=RED_F, lw=1.8, zorder=2)
    
    # Punto hueco en (c, L)
    ax.plot(C, L, 'o', ms=7, mfc='white', mec=RED_F, mew=1.8, zorder=4)

def _zona_prohibida(ax, xlo, xhi):
    """Dibuja las zonas prohibidas (bandas azules)"""
    ylo, yhi = 0.25, 2.75
    ax.axhspan(L + EPS, yhi, alpha=0.35, color=BLUE_Z, zorder=0)
    ax.axhspan(ylo, L - EPS, alpha=0.35, color=BLUE_Z, zorder=0)
    
    mx = (xlo + xhi) / 2
    ax.text(mx, L + EPS + 0.12, 'Zona prohibida',
            ha='center', va='bottom', fontsize=8, color='steelblue', fontweight='bold')
    ax.text(mx, L - EPS - 0.12, 'Zona prohibida',
            ha='center', va='top', fontsize=8, color='steelblue', fontweight='bold')

def _ylabels_L(ax, xlo, xhi):
    """Etiquetas L+ε, L, L-ε a la izquierda"""
    lx = xlo + 0.05 * (xhi - xlo)
    ax.text(lx, L + EPS, 'L+ε', ha='left', va='center',
            fontsize=9, color='steelblue', fontweight='bold')
    ax.text(lx, L - EPS, 'L–ε', ha='left', va='center',
            fontsize=9, color='steelblue', fontweight='bold')

ax.axvspan(C - D, C + D, alpha=0.35, color=GOLD_Z, zorder=0)

# Función completa
xs = np.linspace(xlo, xhi, 800)
ys = _f_unif(xs)
mask = np.abs(xs - C) > 0.015
ax.plot(xs[mask], ys[mask], color=RED_F, lw=1.8, zorder=2)
ax.plot(C, L, 'o', ms=7, mfc='white', mec=RED_F, mew=1.8, zorder=4)

# Etiquetas L a la izquierda con flecha

            arrowprops=dict(arrowstyle='->', color=NAVY, lw=1.5))

# Flechas epsilon (doble sentido)
ex = bx + 0.06 * (xhi - xlo)
ax.annotate('', xy=(ex, L + EPS), xytext=(ex, L),
            arrowprops=dict(arrowstyle='<->', color=RED_F, lw=1.1))

            arrowprops=dict(arrowstyle='<->', color=RED_F, lw=1.1))

        fontsize=10, color=RED_F, fontweight='bold')
ax.text(ex - 0.02 * (xhi - xlo), L - EPS/2, 'ε', ha='right', va='center',
        fontsize=10, color=RED_F, fontweight='bold')

# Etiquetas Y
ax.text(lx, L + EPS, 'L+ε', ha='left', va='center', fontsize=9, color='steelblue', fontweight='bold')

# Puntos de aproximación dorados
for dx in [-0.18, 0.18]:
    ax.plot(C + dx, _f_unif(C + dx), 'o', color=GOLD, ms=7, zorder=3)

# Etiquetas eje X
ax.text(C - D1, 0.30, 'c–δ₁', ha='center', va='bottom', fontsize=7.5, color=GRAY, fontstyle='italic')
ax.text(C - D, 0.30, 'c–δ', ha='center', va='bottom', fontsize=8.5, color=GOLD, fontweight='bold')

# Flechas de intervalo en el eje X (c-δ a c+δ)
ax.annotate('', xy=(C - 0.05, 0.42), xytext=(C - D + 0.05, 0.42),
            arrowprops=dict(arrowstyle='<->', color=RED_F, lw=1.2))
ax.annotate('', xy=(C + D - 0.05, 0.42), xytext=(C + 0.05, 0.42),
            arrowprops=dict(arrowstyle='<->', color=RED_F, lw=1.2))

fig.subplots_adjust(left=0.03, right=0.97, top=0.95, bottom=0.08)
