#| echo: false

import numpy as np
import matplotlib.pyplot as plt

eps = 0.08
x_l = np.linspace(-2*np.pi, -eps, 400)
x_r = np.linspace(eps, 2*np.pi, 400)
y_l = (np.cos(x_l) - 1) / x_l
y_r = (np.cos(x_r) - 1) / x_r

fig, ax = plt.subplots(figsize=(5, 2.8))
ax.grid(True, color="gray", alpha=0.15, linewidth=0.8)
ax.axhline(0, color="black", linewidth=1.8, zorder=2)
ax.axvline(0, color="black", linewidth=1.8, zorder=2)
ax.plot(x_l, y_l, color="#B8860B", linewidth=2.5, zorder=3)
ax.plot(x_r, y_r, color="#B8860B", linewidth=2.5, zorder=3)
ax.plot(0, 0, 'o', color='white', markeredgecolor='#B8860B',
        markeredgewidth=2.2, markersize=9, zorder=5)

ticks = [-2*np.pi, -np.pi, 0, np.pi, 2*np.pi]
labels = [r'$-2\pi$', r'$-\pi$', r'$0$', r'$\pi$', r'$2\pi$']
ax.set_xticks(ticks)
ax.set_xticklabels(labels, fontsize=10)
ax.set_xlim(-2*np.pi - 0.4, 2*np.pi + 0.4)
ax.set_ylim(-1.2, 1.3)
ax.set_yticks([-1, 0, 1])
ax.set_yticklabels(['-1', '0', '1'], fontsize=9)
ax.text(0.98, 0.97, r"$f(\phi)=(\cos(\phi)-1)/\phi$", fontsize=9,
        transform=ax.transAxes, ha='right', va='top')

for spine in ax.spines.values():
    spine.set_visible(False)

plt.tight_layout()
plt.show()
